async function verifyTicket(qrToken) { if (checking) { return; } checking = true; try { const response = await fetch( "/api/verify-ticket?qr_token=" + encodeURIComponent(qrToken) ); const data = await response.json(); if ( data.success && data.valid && data.ticket ) { showResultValid( data.ticket ); return; } if (data.alreadyUsed) { const result = document.getElementById("result"); result.className = "result invalid"; result.innerHTML = `
⚠️

ALREADY USED

This ticket has already been scanned.

`; result.scrollIntoView({ behavior: "smooth" }); return; } showResultInvalid( data.error || "This QR code is not valid." ); } catch (error) { console.error( error ); showResultInvalid( "Unable to verify this ticket. Check your internet connection." ); } }
